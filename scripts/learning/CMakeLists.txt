cmake_minimum_required(VERSION 3.20)
project(learning_preview LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(LVGL "${ROOT}/managed_components/lvgl__lvgl")
set(BOARD "${ROOT}/main/boards/folotoy/ai-passport")
include(CTest)
if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    find_program(NODE_EXECUTABLE node REQUIRED)
    add_executable(check_voice check_voice.cc)
    target_include_directories(check_voice PRIVATE "${BOARD}")
    add_executable(model_trace model_trace.cc)
    target_include_directories(model_trace PRIVATE "${BOARD}")
    add_executable(check_websocket_audio check_websocket_audio.cc)
    target_include_directories(check_websocket_audio PRIVATE "${ROOT}/main/protocols")
    add_test(NAME english_voice COMMAND check_voice)
    add_test(NAME websocket_audio COMMAND check_websocket_audio)
    add_test(NAME chat_transition COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/check_chat_transition.py")
    add_test(NAME prototype_model COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/compare_prototype.py" "$<TARGET_FILE:model_trace>")
endif()
option(BUILD_PREVIEW "Build LVGL preview and audio parser checks (requires managed components)" ON)
if(BUILD_PREVIEW)
    file(GLOB_RECURSE LVGL_SOURCES "${LVGL}/src/*.c")
    add_library(lvgl STATIC ${LVGL_SOURCES})
    target_include_directories(lvgl PUBLIC "${LVGL}")
    target_compile_definitions(lvgl PUBLIC LV_LVGL_H_INCLUDE_SIMPLE=1 LV_CONF_SKIP=1 LV_COLOR_DEPTH=16 LV_USE_OS=0)
    file(GLOB FONTS "${BOARD}/learning_v1_*.c")
    add_executable(learning_preview preview.cc "${BOARD}/learning_render.cc" "${BOARD}/learning_media.cc" ${FONTS})
    target_include_directories(learning_preview PRIVATE "${BOARD}")
    target_link_libraries(learning_preview PRIVATE lvgl)
    add_executable(check_audio check_audio.cc "${BOARD}/learning_media.cc" "${ROOT}/main/audio/demuxer/ogg_demuxer.cc")
    target_include_directories(check_audio PRIVATE "${BOARD}" "${ROOT}/main/audio/demuxer" "${CMAKE_CURRENT_LIST_DIR}/stubs")
    target_link_libraries(check_audio PRIVATE lvgl)
    if(BUILD_TESTING)
        add_test(NAME local_audio COMMAND check_audio)
        add_test(NAME incremental_render COMMAND learning_preview --check-render)
    endif()
endif()
